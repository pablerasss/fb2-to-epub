// PANTALLA COMPLETA function toggleFullscreen() { const content = document.getElementById('reader-content'); const sidebar = document.querySelector('.sidebar'); const header = document.querySelector('.reader-header'); const footer = document.querySelector('.reader-footer'); if (!document.fullscreenElement) { // Entra en pantalla completa document.documentElement.requestFullscreen().catch(err => { alert('No se puede activar pantalla completa'); }); sidebar.style.display = 'none'; header.style.display = 'none'; footer.style.display = 'none'; content.style.padding = '60px 80px'; } else { // Sale de pantalla completa document.exitFullscreen(); sidebar.style.display = 'flex'; header.style.display = 'flex'; footer.style.display = 'flex'; content.style.padding = '40px'; } } // ESC para salir de pantalla completa document.addEventListener('fullscreenchange', () => { if (!document.fullscreenElement) { document.querySelector('.sidebar').style.display = 'flex'; document.querySelector('.reader-header').style.display = 'flex'; document.querySelector('.reader-footer').style.display = 'flex'; } });
